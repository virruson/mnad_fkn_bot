-- Справочник потоков
CREATE TABLE streams (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL 
);

-- Справочник предметов
CREATE TABLE subjects (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL 
);

-- Справочник преподавателей
CREATE TABLE teachers (
    id SERIAL PRIMARY KEY,
    first_name VARCHAR(100),
    last_name VARCHAR(100) NOT NULL,
    middle_name VARCHAR(100), -- отчество
    department VARCHAR(255) -- кафедра
);

-- Справочник типов занятий
CREATE TABLE lesson_types (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE -- Лекция, Семинар, КР
);

-- Основная таблица расписания
CREATE TABLE schedule (
    id SERIAL PRIMARY KEY,
    stream_id INTEGER NOT NULL,
    subject_id INTEGER NOT NULL,
    teacher_id INTEGER NOT NULL,
    lesson_type_id INTEGER NOT NULL,
    lesson_date DATE NOT NULL,
    lesson_time TIME NOT NULL,
    zoom_link TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    -- Внешние ключи
    CONSTRAINT fk_schedule_stream 
        FOREIGN KEY (stream_id) REFERENCES streams(id) ON DELETE CASCADE,
    CONSTRAINT fk_schedule_subject 
        FOREIGN KEY (subject_id) REFERENCES subjects(id) ON DELETE RESTRICT,
    CONSTRAINT fk_schedule_teacher 
        FOREIGN KEY (teacher_id) REFERENCES teachers(id) ON DELETE RESTRICT,
    CONSTRAINT fk_schedule_lesson_type 
        FOREIGN KEY (lesson_type_id) REFERENCES lesson_types(id) ON DELETE RESTRICT
);

-- Индексы для оптимизации запросов
CREATE INDEX idx_schedule_date ON schedule(lesson_date);
CREATE INDEX idx_schedule_stream_date ON schedule(stream_id, lesson_date);
CREATE INDEX idx_schedule_teacher_date ON schedule(teacher_id, lesson_date);
CREATE INDEX idx_teachers_last_name ON teachers(last_name);
CREATE INDEX idx_subjects_name ON subjects(name);
CREATE INDEX idx_streams_name ON streams(name);

SELECT 
    to_char(s.lesson_date, 'DD.MM.YYYY') as date,
    to_char(s.lesson_time, 'HH24:MI') as time,
    sub.name as subject,
    t.last_name || ' ' || t.first_name as teacher,
    lt.name as lesson_type,
    st.name as stream,
    s.zoom_link
FROM schedule s
JOIN subjects sub ON s.subject_id = sub.id
JOIN teachers t ON s.teacher_id = t.id
JOIN lesson_types lt ON s.lesson_type_id = lt.id
JOIN streams st ON s.stream_id = st.id
WHERE s.lesson_date >= CURRENT_DATE
ORDER BY s.lesson_date, s.lesson_time;
